import { Chart } from '@core/Chart';
import { Grid } from '@components/Grid';
import { Tooltip } from '@components/Tooltip';
import { Axis } from '@components/Axis';
import type { IDataPoint, IChartOptions } from '@_types/data';
import { scaleLinear } from '@utils/math';

interface IInteractivePoint {
  item: IDataPoint;
  x: number;
  y: number;
  color: string;
}

export class LineChart extends Chart {
  private gridComponent: Grid;
  private tooltipComponent: Tooltip;
  private axisComponent: Axis;
  private interactivePoints: IInteractivePoint[] = [];
  private maxPointsThreshold: number;

  constructor(containerId: string, data: IDataPoint[], options: IChartOptions = {}) {
    super(containerId, data, options);
    this.gridComponent = new Grid({ tickCount: 5 });
    this.tooltipComponent = new Tooltip(this.container);
    this.axisComponent = new Axis({ tickCount: 5 });
    this.maxPointsThreshold = this.options.maxRenderPoints;

    this.setupInteractions();
  }

  private setupInteractions(): void {
    this.canvasManager.element.addEventListener('mousemove', (e: MouseEvent) => {
      if (!Array.isArray(this.data) || this.data.length === 0) return;

      // Map clean vectors instantly via the central canvas manager
      const mouse = this.canvasManager.getMouseCoordinates(e);

      const hoverRadius = 15;
      const hit = this.interactivePoints.find((pt) => {
        const dx = mouse.x - pt.x;
        const dy = mouse.y - pt.y;
        return Math.sqrt(dx * dx + dy * dy) <= hoverRadius;
      });

      if (hit) {
        this.tooltipComponent.show({ x: hit.x, y: hit.y }, {
          title: hit.item.label,
          value: hit.item.value,
          color: hit.color
        }, this.theme);
        this.canvasManager.element.style.cursor = 'pointer';
      } else {
        this.tooltipComponent.hide();
        this.canvasManager.element.style.cursor = 'default';
      }
    });

    this.canvasManager.element.addEventListener('mouseleave', () => {
      this.tooltipComponent.hide();
      this.canvasManager.element.style.cursor = 'default';
    });
  }

  private downsampleData(rawSrc: IDataPoint[], targetBucketSize: number): IDataPoint[] {
    if (rawSrc.length <= targetBucketSize) return rawSrc;

    const sampled: IDataPoint[] = [];
    sampled.push(rawSrc[0]); 

    const bucketWidth = (rawSrc.length - 2) / (targetBucketSize - 2);

    for (let i = 0; i < targetBucketSize - 2; i++) {
      const startIdx = Math.floor(i * bucketWidth + 1);
      const endIdx = Math.floor((i + 1) * bucketWidth + 1);

      let maxVal = -Infinity;
      let minVal = Infinity;
      let maxObj = rawSrc[startIdx];
      let minObj = rawSrc[startIdx];

      for (let j = startIdx; j < endIdx; j++) {
        if (j >= rawSrc.length) break;
        const curr = rawSrc[j];
        if (curr.value > maxVal) { maxVal = curr.value; maxObj = curr; }
        if (curr.value < minVal) { minVal = curr.value; minObj = curr; }
      }

      if (rawSrc.indexOf(minObj) < rawSrc.indexOf(maxObj)) {
        sampled.push(minObj);
        sampled.push(maxObj);
      } else {
        sampled.push(maxObj);
        sampled.push(minObj);
      }
    }

    sampled.push(rawSrc[rawSrc.length - 1]);
    return sampled;
  }

  public render(): void {
    this.clear();
    this.interactivePoints = [];

    if (!Array.isArray(this.data) || this.data.length === 0) return;

    const renderData = this.downsampleData(this.data as unknown as IDataPoint[], this.maxPointsThreshold);
    const maxValue = Math.max(...renderData.map((d) => d.value), 0);
    const totalItems = renderData.length;

    this.gridComponent.draw(this.ctx, this.margin, this.chartWidth, this.chartHeight, maxValue, this.theme);
    this.axisComponent.drawY(this.ctx, this.margin, this.chartHeight, maxValue, this.theme);

    const xScale = scaleLinear([0, totalItems - 1], [this.margin.left, this.margin.left + this.chartWidth]);
    const yScale = scaleLinear([0, maxValue], [this.margin.top + this.chartHeight, this.margin.top]);
    const strokeColor = this.theme.fallbackColors[0];

    this.ctx.save();
    this.ctx.beginPath();
    this.ctx.lineWidth = 2.5;
    this.ctx.strokeStyle = strokeColor;
    this.ctx.lineJoin = 'round';
    this.ctx.lineCap = 'round';

    for (let i = 0; i < totalItems; i++) {
      const x = xScale(i);
      const y = yScale(renderData[i].value);

      if (i === 0) {
        this.ctx.moveTo(x, y);
      } else {
        this.ctx.lineTo(x, y);
      }

      this.interactivePoints.push({
        item: renderData[i],
        x,
        y,
        color: strokeColor
      });
    }
    this.ctx.stroke();

    if (totalItems <= 30) {
      for (const pt of this.interactivePoints) {
        this.ctx.beginPath();
        this.ctx.arc(pt.x, pt.y, 4, 0, 2 * Math.PI);
        this.ctx.fillStyle = this.theme.background;
        this.ctx.fill();
        this.ctx.strokeStyle = pt.color;
        this.ctx.lineWidth = 2;
        this.ctx.stroke();
      }
    }

    const stepSkip = Math.ceil(totalItems / 6);
    for (let i = 0; i < totalItems; i += stepSkip) {
      this.axisComponent.drawXLabel(
        this.ctx,
        renderData[i].label,
        xScale(i),
        this.margin.top + this.chartHeight + 8,
        this.theme
      );
    }

    this.axisComponent.drawXBaseline(this.ctx, this.margin, this.chartWidth, this.chartHeight, this.theme);
    this.ctx.restore();
  }
}
