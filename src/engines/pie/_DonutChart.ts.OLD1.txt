import { Chart } from '@core/Chart';
import { Tooltip } from '@components/Tooltip';
import type { IDataPoint, IChartOptions } from '@_types/data';
import { computePieSlices, polarToCartesian, type IPieSliceData } from './PieMath';

export class DonutChart extends Chart {
  private tooltipComponent: Tooltip;
  private sliceInteractiveZones: { slice: IPieSliceData; color: string }[] = [];
  private innerRadiusRatio = 0.6; 

  constructor(containerId: string, data: IDataPoint[], options: IChartOptions = {}) {
    super(containerId, data, options);
    this.tooltipComponent = new Tooltip(this.container);
    this.setupInteractions();
  }

  private setupInteractions(): void {
    this.canvasManager.element.addEventListener('mousemove', (e: MouseEvent) => {
      if (!Array.isArray(this.data) || this.data.length === 0) return;

      const mouse = this.canvasManager.getMouseCoordinates(e);

      const centerX = this.margin.left + this.chartWidth / 2;
      const centerY = this.margin.top + this.chartHeight / 2;
      const outerRadius = (Math.min(this.chartWidth, this.chartHeight) / 2) * 0.85;
      const innerRadius = outerRadius * this.innerRadiusRatio;

      const dx = mouse.x - centerX;
      const dy = mouse.y - centerY;
      const distance = Math.sqrt(dx * dx + dy * dy);

      if (distance > outerRadius || distance < innerRadius) {
        this.tooltipComponent.hide();
        this.canvasManager.element.style.cursor = 'default';
        return;
      }

      let hoverAngle = Math.atan2(dy, dx);
      if (hoverAngle < -Math.PI / 2) {
        hoverAngle += 2 * Math.PI;
      }

      const hit = this.sliceInteractiveZones.find((zone) => {
        return hoverAngle >= zone.slice.startAngle && hoverAngle <= zone.slice.endAngle;
      });

      if (hit) {
        this.tooltipComponent.show({ x: mouse.x, y: mouse.y }, {
          title: hit.slice.label,
          value: hit.slice.value,
          color: hit.color
        }, this.theme);
        this.canvasManager.element.style.cursor = 'pointer';
      } else {
        this.tooltipComponent.hide();
        this.canvasManager.element.style.cursor = 'default';
      }
    });

    this.canvasManager.element.addEventListener('mouseleave', () => {
      this.tooltipComponent.hide();
      this.canvasManager.element.style.cursor = 'default';
    });
  }

  public render(): void {
    this.clear();
    this.sliceInteractiveZones = [];

    if (!Array.isArray(this.data) || this.data.length === 0) return;

    const centerX = this.margin.left + this.chartWidth / 2;
    const centerY = this.margin.top + this.chartHeight / 2;
    const outerRadius = (Math.min(this.chartWidth, this.chartHeight) / 2) * 0.85;
    const innerRadius = outerRadius * this.innerRadiusRatio;

    const slices = computePieSlices(this.data);

    slices.forEach((slice, index) => {
      const sliceColor = this.theme.fallbackColors[index % this.theme.fallbackColors.length];
      
      this.ctx.beginPath();
      this.ctx.arc(centerX, centerY, outerRadius, slice.startAngle, slice.endAngle, false);
      this.ctx.arc(centerX, centerY, innerRadius, slice.endAngle, slice.startAngle, true);
      this.ctx.closePath();

      this.ctx.fillStyle = sliceColor;
      this.ctx.fill();

      this.ctx.strokeStyle = this.theme.background;
      this.ctx.lineWidth = 2;
      this.ctx.stroke();

      this.sliceInteractiveZones.push({ slice, color: sliceColor });
    });

    slices.forEach((slice) => {
      if (slice.percentage > 0.05) {
        const middleAngle = slice.startAngle + (slice.endAngle - slice.startAngle) / 2;
        const labelRadius = innerRadius + (outerRadius - innerRadius) / 2;
        const textPos = polarToCartesian(centerX, centerY, labelRadius, middleAngle);

        this.ctx.save();
        this.ctx.fillStyle = '#ffffff';
        this.ctx.font = 'bold 11px sans-serif';
        this.ctx.textAlign = 'center';
        this.ctx.textBaseline = 'middle';
        
        const displayPercent = `${(slice.percentage * 100).toFixed(0)}%`;
        this.ctx.fillText(displayPercent, textPos.x, textPos.y);
        this.ctx.restore();
      }
    });

    const totalSum = this.data.reduce((sum, item) => sum + item.value, 0);
    this.ctx.save();
    this.ctx.fillStyle = this.theme.textPrimary;
    this.ctx.font = 'bold 20px sans-serif';
    this.ctx.textAlign = 'center';
    this.ctx.textBaseline = 'middle';
    this.ctx.fillText(totalSum.toLocaleString(), centerX, centerY);
    this.ctx.restore();
  }
}
